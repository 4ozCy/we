local function Tween(Object, Properties, Duration, Style, Direction, Delay)
    local Info = TweenInfo.new(
        Duration or 0.2,
        Style or Enum.EasingStyle.Quint,
        Direction or Enum.EasingDirection.Out,
        0, false, Delay or 0
    )
    local Animation = TweenService:Create(Object, Info, Properties)
    Animation:Play()
    return Animation
end

local function CreateESP(Player)
    if ESPObjects[Player] then return end
    local Data = {
        Lines = {},
        HealthBar = CreateLine(),
        HealthBackground = CreateLine(),
        Username = NewDrawing("Text"),
        Distance = NewDrawing("Text"),
        HealthText = NewDrawing("Text"),
    }
    for i = 1, 8 do
        Data.Lines[i] = CreateLine()
    end
    Data.Username.Size = 13
    Data.Username.Center = true
    Data.Username.Outline = true
    Data.Username.OutlineColor = Color3.fromRGB(0, 0, 0)
    Data.Username.Color = Settings.TextColor
    Data.Username.Visible = false
    Data.Distance.Size = 12
    Data.Distance.Center = true
    Data.Distance.Outline = true
    Data.Distance.OutlineColor = Color3.fromRGB(0, 0, 0)
    Data.Distance.Color = Settings.TextColor
    Data.Distance.Visible = false
    Data.HealthText.Size = 12
    Data.HealthText.Center = true
    Data.HealthText.Outline = true
    Data.HealthText.OutlineColor = Color3.fromRGB(0, 0, 0)
    Data.HealthText.Color = Settings.TextColor
    Data.HealthText.Visible = false
    ESPObjects[Player] = Data
end

local function HideESP(Data)
    for _, Line in ipairs(Data.Lines) do
        Line.Visible = false
    end
    Data.HealthBar.Visible = false
    Data.HealthBackground.Visible = false
    Data.Username.Visible = false
    Data.Distance.Visible = false
    Data.HealthText.Visible = false
end

local function RemoveESP(Player)
    local Data = ESPObjects[Player]
    if not Data then return end
    for _, Line in ipairs(Data.Lines) do
        pcall(function() Line:Remove() end)
    end
    pcall(function()
        Data.HealthBar:Remove()
        Data.HealthBackground:Remove()
        Data.Username:Remove()
        Data.Distance:Remove()
        Data.HealthText:Remove()
    end)
    ESPObjects[Player] = nil
end

local function GetPlayerColor(Player)
    if Settings.TeamColor and Player.TeamColor then
        return Player.TeamColor.Color
    end
    return Settings.DefaultColor
end

local function GetBoundingBox(Character)
    local Parts = {}
    for _, Object in ipairs(Character:GetChildren()) do
        if Object:IsA("BasePart") then
            table.insert(Parts, Object)
        end
    end
    if #Parts == 0 then return nil end
    local MinX, MinY = math.huge, math.huge
    local MaxX, MaxY = -math.huge, -math.huge
    local Visible = false
    for _, Part in ipairs(Parts) do
        local Size = Part.Size
        local CF = Part.CFrame
        local Corners = {
            CF * Vector3.new(-Size.X / 2, -Size.Y / 2, -Size.Z / 2),
            CF * Vector3.new(-Size.X / 2, -Size.Y / 2, Size.Z / 2),
            CF * Vector3.new(-Size.X / 2, Size.Y / 2, -Size.Z / 2),
            CF * Vector3.new(-Size.X / 2, Size.Y / 2, Size.Z / 2),
            CF * Vector3.new(Size.X / 2, -Size.Y / 2, -Size.Z / 2),
            CF * Vector3.new(Size.X / 2, -Size.Y / 2, Size.Z / 2),
            CF * Vector3.new(Size.X / 2, Size.Y / 2, -Size.Z / 2),
            CF * Vector3.new(Size.X / 2, Size.Y / 2, Size.Z / 2),
        }
        for _, Corner in ipairs(Corners) do
            local ScreenPosition, OnScreen = Camera:WorldToViewportPoint(Corner)
            if OnScreen and ScreenPosition.Z > 0 then
                Visible = true
            end
            if ScreenPosition.Z > 0 then
                MinX = math.min(MinX, ScreenPosition.X)
                MinY = math.min(MinY, ScreenPosition.Y)
                MaxX = math.max(MaxX, ScreenPosition.X)
                MaxY = math.max(MaxY, ScreenPosition.Y)
            end
        end
    end
    if not Visible or MinX == math.huge then return nil end
    return MinX, MinY, MaxX, MaxY
end

local function UpdateCornerBox(Data, X, Y, Width, Height, Color)
    local CX = X + Width
    local CY = Y + Height
    local Horizontal = math.max(Width * Settings.CornerLength, 8)
    local Vertical = math.max(Height * Settings.CornerLength, 8)
    local Lines = Data.Lines
    Lines[1].From = Vector2.new(X, Y)
    Lines[1].To = Vector2.new(X + Horizontal, Y)
    Lines[2].From = Vector2.new(X, Y)
    Lines[2].To = Vector2.new(X, Y + Vertical)
    Lines[3].From = Vector2.new(CX, Y)
    Lines[3].To = Vector2.new(CX - Horizontal, Y)
    Lines[4].From = Vector2.new(CX, Y)
    Lines[4].To = Vector2.new(CX, Y + Vertical)
    Lines[5].From = Vector2.new(X, CY)
    Lines[5].To = Vector2.new(X + Horizontal, CY)
    Lines[6].From = Vector2.new(X, CY)
    Lines[6].To = Vector2.new(X, CY - Vertical)
    Lines[7].From = Vector2.new(CX, CY)
    Lines[7].To = Vector2.new(CX - Horizontal, CY)
    Lines[8].From = Vector2.new(CX, CY)
    Lines[8].To = Vector2.new(CX, CY - Vertical)
    for _, Line in ipairs(Lines) do
        Line.Color = Color
        Line.Thickness = Settings.BoxThickness
        Line.Visible = true
    end
end

local function UpdatePlayerESP(Player, Data)
    if not Settings.ESPEnabled or Player == LocalPlayer then
        HideESP(Data)
        return
    end
    if Settings.TeamCheck and LocalPlayer.Team and Player.Team == LocalPlayer.Team then
        HideESP(Data)
        return
    end
    local Character = Player.Character
    if not Character then
        HideESP(Data)
        return
    end
    local Humanoid = Character:FindFirstChildOfClass("Humanoid")
    local Root = Character:FindFirstChild("HumanoidRootPart")
    if not Humanoid or not Root or Humanoid.Health <= 0 then
        HideESP(Data)
        return
    end
    local LocalCharacter = LocalPlayer.Character
    local LocalRoot = LocalCharacter and LocalCharacter:FindFirstChild("HumanoidRootPart")
    if not LocalRoot then
        HideESP(Data)
        return
    end
    local Distance = (Root.Position - LocalRoot.Position).Magnitude
    if Distance > Settings.MaxDistance then
        HideESP(Data)
        return
    end
    local X1, Y1, X2, Y2 = GetBoundingBox(Character)
    if not X1 then
        HideESP(Data)
        return
    end
    local Width = X2 - X1
    local Height = Y2 - Y1
    if Width <= 1 or Height <= 1 then
        HideESP(Data)
        return
    end
    local Color = GetPlayerColor(Player)
    UpdateCornerBox(Data, X1, Y1, Width, Height, Color)
    local HealthPercent = math.clamp(Humanoid.Health / math.max(Humanoid.MaxHealth, 1), 0, 1)
    local HealthHeight = Height * HealthPercent
    Data.HealthBackground.From = Vector2.new(X1 - 6, Y1)
    Data.HealthBackground.To = Vector2.new(X1 - 6, Y2)
    Data.HealthBackground.Color = Color3.fromRGB(30, 30, 35)
    Data.HealthBackground.Thickness = 4
    Data.HealthBackground.Visible = true
    Data.HealthBar.From = Vector2.new(X1 - 6, Y2)
    Data.HealthBar.To = Vector2.new(X1 - 6, Y2 - HealthHeight)
    Data.HealthBar.Color = Settings.HealthColor and Color3.fromRGB(
        math.floor(255 * (1 - HealthPercent)),
        math.floor(255 * HealthPercent),
        60
    ) or Color3.fromRGB(100, 255, 150)
    Data.HealthBar.Thickness = 2
    Data.HealthBar.Visible = true
    Data.Username.Text = Player.DisplayName .. "  @" .. Player.Name
    Data.Username.Position = Vector2.new(X1 + Width / 2, Y1 - 18)
    Data.Username.Visible = true
    Data.Distance.Text = string.format("%dm", math.floor(Distance))
    Data.Distance.Position = Vector2.new(X1 + Width / 2, Y2 + 5)
    Data.Distance.Visible = true
    Data.HealthText.Text = string.format("%d%%", math.floor(HealthPercent * 100))
    Data.HealthText.Position = Vector2.new(X1 - 28, Y1 + Height / 2)
    Data.HealthText.Visible = true
end

local function Show(Name)
    if CurrentPage == Name then return end
    local OldName = CurrentPage
    if OldName and Pages[OldName] then
        Tween(Pages[OldName].Group, {GroupTransparency = 1}, 0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
        task.delay(0.13, function()
            if CurrentPage ~= OldName and Pages[OldName] then
                Pages[OldName].Group.Visible = false
            end
        end)
    end
    local Target = Pages[Name]
    Target.Group.Visible = true
    Target.Group.GroupTransparency = 1
    Target.Frame.Position = UDim2.new(0, 15, 0, 26)
    Tween(Target.Group, {GroupTransparency = 0}, 0.28)
    Tween(Target.Frame, {Position = UDim2.new(0, 15, 0, 15)}, 0.34)
    for N, Data in pairs(Tabs) do
        local Active = N == Name
        if Active then
            Tween(Data.Button, {BackgroundColor3 = Theme.Surface3, BackgroundTransparency = 0.15}, 0.25)
            Data.Active.Visible = true
            Data.Active.Size = UDim2.new(0, 3, 0, 0)
            Tween(Data.Active, {Size = UDim2.new(0, 3, 0, 30)}, 0.32, Enum.EasingStyle.Back)
            Tween(Data.Icon, {ImageColor3 = Theme.Text}, 0.2)
            Tween(Data.Title, {TextColor3 = Theme.Text}, 0.2)
            Tween(Data.Desc, {TextColor3 = Theme.Muted}, 0.2)
        else
            Tween(Data.Button, {BackgroundColor3 = Theme.Surface2, BackgroundTransparency = 0.55}, 0.25)
            Tween(Data.Active, {Size = UDim2.new(0, 3, 0, 0)}, 0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
            task.delay(0.17, function()
                if CurrentPage ~= N then
                    Data.Active.Visible = false
                end
            end)
            Tween(Data.Icon, {ImageColor3 = Theme.Muted}, 0.2)
            Tween(Data.Title, {TextColor3 = Color3.fromRGB(188, 189, 199)}, 0.2)
            Tween(Data.Desc, {TextColor3 = Theme.Dim}, 0.2)
        end
    end
    CurrentPage = Name
end

local function Slider(PageObject, Title, Description, Min, Max, Step, Get, Set, Format, Asset)

    BarArea.Parent = Holder
    local Bar = Instance.new("Frame")
    Bar.Size = UDim2.new(1, 0, 0, 5)
    Bar.Position = UDim2.new(0, 0, 0.5, -2)
    Bar.BackgroundColor3 = Color3.fromRGB(48, 49, 60)
    Bar.BorderSizePixel = 0
    Bar.Parent = BarArea
    Corner(Bar, 3)
    local Fill = Instance.new("Frame")
    Fill.Size = UDim2.new(0, 0, 1, 0)
    Fill.BackgroundColor3 = Theme.Accent
    Fill.BorderSizePixel = 0
    Fill.Parent = Bar
    Corner(Fill, 3)
    AccentGradient(Fill, 0)
    local Knob = Instance.new("Frame")
    Knob.Size = UDim2.new(0, 14, 0, 14)
    Knob.AnchorPoint = Vector2.new(0.5, 0.5)
    Knob.Position = UDim2.new(0, 0, 0.5, 0)
    Knob.BackgroundColor3 = Theme.Text
    Knob.BorderSizePixel = 0
    Knob.Parent = Bar
    Corner(Knob, 7)
    local KnobStroke = Border(Knob, 0.7, 1.5)
    Holder.MouseEnter:Connect(function()
        Tween(Holder, {BackgroundTransparency = 0.1}, 0.18)
    end)
    Holder.MouseLeave:Connect(function()
        Tween(Holder, {BackgroundTransparency = 0.22}, 0.18)
    end)
    local Dragging = false
    local function Refresh(Animate)
        local V = Get()
        local P = math.clamp((V - Min) / (Max - Min), 0, 1)
        Value.Text = Format and Format(V) or tostring(V)
        if Animate then
            Tween(Fill, {Size = UDim2.new(P, 0, 1, 0)}, 0.32)
            Tween(Knob, {Position = UDim2.new(P, 0, 0.5, 0)}, 0.32)
        else
            Fill.Size = UDim2.new(P, 0, 1, 0)
            Knob.Position = UDim2.new(P, 0, 0.5, 0)
        end
    end
    local function Update(Input)
        local P = math.clamp((Input.Position.X - Bar.AbsolutePosition.X) / Bar.AbsoluteSize.X, 0, 1)
        local Raw = Min + (Max - Min) * P
        local V = math.floor(Raw / Step + 0.5) * Step
        Set(math.clamp(V, Min, Max))
        Refresh(false)
    end
    BarArea.InputBegan:Connect(function(Input)
        if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
            Dragging = true
            Update(Input)
            Tween(Knob, {Size = UDim2.new(0, 18, 0, 18)}, 0.18, Enum.EasingStyle.Back)
            Tween(KnobStroke, {Transparency = 0.25, Color = Theme.Accent}, 0.18)
        end
    end)
    Track(UserInputService.InputChanged:Connect(function(Input)
        if Dragging and (Input.UserInputType == Enum.UserInputType.MouseMovement or Input.UserInputType == Enum.UserInputType.Touch) then
            Update(Input)
        end
    end))
    Track(UserInputService.InputEnded:Connect(function(Input)
        if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
            if Dragging then
                Dragging = false
                Tween(Knob, {Size = UDim2.new(0, 14, 0, 14)}, 0.2, Enum.EasingStyle.Back)
                Tween(KnobStroke, {Transparency = 0.7, Color = Theme.Border}, 0.2)
            end
        end
    end))
    Refresh(true)
    table.insert(ControlRegistry, Refresh)
    return Holder
end

local function Alive(Player)
    local Character = Player and Player.Character
    local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
    return Humanoid and Humanoid.Health > 0
end

local function ClosestTarget()
    local Best, BestDistance
    local Center = Camera.ViewportSize / 2
    for _, Player in ipairs(Players:GetPlayers()) do
        if Player ~= LocalPlayer and Alive(Player) then
            if not AimbotTeamCheck or Player.Team ~= LocalPlayer.Team then
                local Head = Player.Character:FindFirstChild("Head")
                if Head then
                    local Position, Visible = Camera:WorldToViewportPoint(Head.Position)
                    if Visible and Position.Z > 0 then
                        local Distance = (Vector2.new(Position.X, Position.Y) - Center).Magnitude
                        if Distance <= AimbotFOV and (not BestDistance or Distance < BestDistance) then
                            Best, BestDistance = Player, Distance
                        end
                    end
                end
            end
        end
    end
    return Best
end

local function UpdateTouchFling(State)
    TouchFlingEnabled = State
    if State then
        local plr = Players.LocalPlayer
        local rS = game:GetService("RunService")
        local mov = 0.1
        TouchFlingThread = coroutine.create(function()
            while TouchFlingEnabled do
                rS.Heartbeat:Wait()
                local hrt = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
                if hrt then
                    local vel = hrt.Velocity
                    hrt.Velocity = vel * 1e15 + Vector3.new(0, 1e15, 0)
                    rS.RenderStepped:Wait()
                    hrt.Velocity = vel
                    rS.Stepped:Wait()
                    hrt.Velocity = vel + Vector3.new(0, mov, 0)
                    mov = -mov
                end
            end
        end)
        coroutine.resume(TouchFlingThread)
    else
        if TouchFlingThread then
            coroutine.close(TouchFlingThread)
            TouchFlingThread = nil
        end
        local hrt = Players.LocalPlayer.Character and Players.LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if hrt then
            hrt.Velocity = Vector3.new(0, 0, 0)
        end
    end
end

    for _, Player in ipairs(Players:GetPlayers()) do if Player ~= LocalPlayer and Player.DisplayName:lower() == Input then return Player end end
    for _, Player in ipairs(Players:GetPlayers()) do if Player ~= LocalPlayer and Player.Name:lower():sub(1, #Input) == Input then return Player end end

local function SetHubVisible(Visible)
    if Visible then
        Main.Visible = true
        Open.Visible = false
        Tween(Main, {GroupTransparency = 0}, 0.35, Enum.EasingStyle.Quart)
        Tween(Main, {Size = UDim2.new(0, 680, 0, 470)}, 0.4, Enum.EasingStyle.Quart)
    else
        Tween(Main, {GroupTransparency = 1}, 0.22, Enum.EasingStyle.Quart, Enum.EasingDirection.In)
        Tween(Main, {Size = UDim2.new(0, 646, 0, 444)}, 0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.In)
        task.delay(0.26, function()
            Main.Visible = false
            Open.Visible = true
            Open.Size = UDim2.new(0, 96, 0, 36)
            Open.BackgroundTransparency = 0.3
            Tween(Open, {Size = UDim2.new(0, 118, 0, 44), BackgroundTransparency = 0.08}, 0.32, Enum.EasingStyle.Back)
        end)
    end
end

local Dragging = false
local DragStart
local StartPosition
Header.InputBegan:Connect(function(Input)
    if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
        Dragging = true
        DragStart = Input.Position
        StartPosition = Main.Position
    end
end)
Track(UserInputService.InputChanged:Connect(function(Input)
    if Dragging and (Input.UserInputType == Enum.UserInputType.MouseMovement or Input.UserInputType == Enum.UserInputType.Touch) then
        local Delta = Input.Position - DragStart
        Main.Position = UDim2.new(
            StartPosition.X.Scale, StartPosition.X.Offset + Delta.X,
            StartPosition.Y.Scale, StartPosition.Y.Offset + Delta.Y
        )
    end
end))
Track(UserInputService.InputEnded:Connect(function(Input)
    if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
        Dragging = false
    end
end))

Track(RunService.RenderStepped:Connect(function()
    Camera = workspace.CurrentCamera
    if not Camera then return end
    
    if SpectateEnabled and SpectateName ~= "" then
        local Target = Players:FindFirstChild(SpectateName)
        if Target and Target.Character and Target.Character:FindFirstChild("Humanoid") then
            Camera.CameraSubject = Target.Character:FindFirstChild("Humanoid")
        end
    else
        local LocalHumanoid = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid")
        if LocalHumanoid and Camera.CameraSubject ~= LocalHumanoid then
            Camera.CameraSubject = LocalHumanoid
        end
    end

    FOVCircle.Position = Camera.ViewportSize / 2
    FOVCircle.Radius = AimbotFOV
    FOVCircle.Visible = AimbotEnabled
    
    if AimbotEnabled and not SpectateEnabled then
        if not AimbotTarget or not Alive(AimbotTarget) then
            AimbotTarget = ClosestTarget()
        end
        if AimbotTarget and AimbotTarget.Character then
            local Head = AimbotTarget.Character:FindFirstChild("Head")
            if Head then
                local Goal = CFrame.new(Camera.CFrame.Position, Head.Position)
                Camera.CFrame = Camera.CFrame:Lerp(Goal, 1 / math.max(AimbotSmoothness, 1))
            end
        end
    end
    
    for Player, Data in pairs(ESPObjects) do
        local Ok = pcall(UpdatePlayerESP, Player, Data)
        if not Ok then
            pcall(HideESP, Data)
        end
    end
end))

Track(RunService.Heartbeat:Connect(function(Delta)
    if AntiFlingEnabled then
        local Character = LocalPlayer.Character
        local Root = Character and Character:FindFirstChild("HumanoidRootPart")
        if Root and Root.Velocity.Magnitude > 500 then
            Root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            Root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
        end
    end

    if not TPWalkEnabled then return end
    local Character = LocalPlayer.Character
    local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
    local Root = Character and Character:FindFirstChild("HumanoidRootPart")
    if Humanoid and Root and Humanoid.MoveDirection.Magnitude > 0 then
        Character:PivotTo(Root.CFrame + Humanoid.MoveDirection * Delta * TPWalkSpeed * 10)
    end
end))
